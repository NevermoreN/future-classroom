```sh
CC_i686_win7_windows_gnu=i686-w64-mingw32-gcc AR_i686_win7_windows_gnu=i686-w64-mingw32-ar cargo +nightly-2026-10-29 build -p fc-bootstrap --release --target i686-win7-windows-gnu -Z build-std=std,panic_unwind --locked
```
